10static inline void getModelMatrix(vec2 pos, vec2 size, float rotation, mat4 model)
11{

13 glm_translate(model, (vec3) { 0.5f * size[0], 0.5f * size[1], 0.0f }); // move origin to center of quad
14 glm_rotate(model, glm_rad(rotation), (vec3) { 0.0f, 0.0f, 1.0f }); // rotate over z-axis
15 glm_translate(model, (vec3) { -0.5f * size[0], -0.5f * size[1], 0.0f }); // reset origin
